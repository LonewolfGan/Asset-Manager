    expect(calculateStrength('aaaaaaaa')).toBe('weak')
  })
  it('strength: 16 chars mixed = strong', () => {

1d. Translation completeness
javascript// tests/unit/i18n/translation-completeness.test.js
import en from '@/locales/en.json'
import fr from '@/locales/fr.json'
import ar from '@/locales/ar.json'

const flattenKeys = (obj, prefix = '') =>
  Object.keys(obj).reduce((acc, key) => {
    const fullKey = prefix ? `${prefix}.${key}` : key
    if (typeof obj[key] === 'object' && obj[key] !== null) {
      Object.assign(acc, flattenKeys(obj[key], fullKey))
    } else {
      acc[fullKey] = obj[key]
    }
    return acc
  }, {})

describe('Translation Completeness', () => {
  const enKeys = Object.keys(flattenKeys(en))
  const frKeys = Object.keys(flattenKeys(fr))
  const arKeys = Object.keys(flattenKeys(ar))

  it('FR has all EN keys', () => {
    const missing = enKeys.filter(k => !frKeys.includes(k))
    expect(missing).toEqual([]) // will list missing keys if fails
  })
  it('AR has all EN keys', () => {
    const missing = enKeys.filter(k => !arKeys.includes(k))
    expect(missing).toEqual([])
  })
  it('no translation value is empty string', () => {
    const flat = flattenKeys(en)
    const empty = Object.entries(flat).filter(([, v]) => v === '')
    expect(empty).toEqual([])
  })
  it('no untranslated marker left in any language file', () => {
    const flatFr = flattenKeys(fr)
    const marked = Object.entries(flatFr).filter(([, v]) => String(v).startsWith('⚠️'))
    expect(marked).toEqual([]) // fail = translation was marked but never translated
  })
})
